# The approved command surface of the `forklift` binary.
#
# The surface check (crates/forklift/src/surface_check.rs) fails until this file and the binary
# name the same commands, so a new command is a decision made here, on review, not a side effect
# of adding a variant to cli.rs. Adding one means placing it in [commands] and classifying it
# against the definition of each list below; removing or renaming one means editing it here too.
# The coverage map beside this file (coverage.txt) records what every earlier command became.

[daily-loop]
# The commands a person runs every day, kept top-level and short. The surface being built gives

# rule, and no check holds it until the change that removes its last violation.

[destructive]
# An operation is destructive when it destroys working-tree content, history or a verified trust
# guarantee that exists now, and no Forklift command can bring it back from this warehouse alone.
# When only one mode of a command is destructive, the entry names the flag and value selecting it.
restore
scope-prune
office regenesis
office accept-regenesis
lower --keep …=theirs

[recorded-text]
# The commands that record text: free-form prose a person writes to describe a record (a
# parcel's description, a note, a comment, a review, a tag or delivery message, a haul's longer
# description). A keyed value (manifest set VALUE, config set VALUE) and a short title (haul open --title)
# are not recorded text.
stack
pick
deliver
haul open
haul comment
haul review
manifest note
manifest approve
manifest provenance
